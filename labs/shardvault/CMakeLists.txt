cmake_minimum_required(VERSION 3.20)
project(shardvault LANGUAGES CXX)
add_executable(shardvault src/main.cpp)
target_compile_features(shardvault PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(shardvault PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(shardvault PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
